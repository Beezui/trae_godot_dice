```
NewDiceDemo (Node3D)
├─ DiceManager (Node3D, dice_manager.gd)
│  └─ 设置参数:
│     - dice_count: 3
│     - current_scene: "1003"
├─ UI (CanvasLayer)
│  ├─ ChargeLabel (Label)
│  └─ ResultLabel (Label)
└─ Camera (Camera3D)
```

### 步骤 3: 编写场景脚本
```gdscript
# scenes/new_dice_demo.gd
extends Node3D

@onready var dice_manager = $DiceManager
@onready var charge_label = $UI/ChargeLabel
@onready var result_label = $UI/ResultLabel

var is_charging = false
var charge_start_time = 0.0

func _ready():
    # DiceManager 会自动加载配置并应用贴图
    print("新场景初始化完成")

func _process(delta):
    if is_charging:
        var charge_ratio = DiceThrowController.get_instance().update_charge(delta)
        charge_label.text = "蓄力：%d%%" % (charge_ratio * 100)

func _input(event):
    if event.is_action_pressed("ui_space"):
        if not is_charging:
            # 开始蓄力
            is_charging = true
            DiceThrowController.get_instance().start_charge()
            charge_label.text = "蓄力：0%"
    
    elif event.is_action_released("ui_space"):
        if is_charging:
            # 结束蓄力并投掷
            is_charging = false
            var charge_ratio = DiceThrowController.get_instance().get_charge_ratio()
            DiceThrowController.get_instance().end_charge(dice_manager.dice_instances)
            charge_label.text = ""

func update_result_display(values: Array):
    result_label.text = "结果：%s" % str(values)
```

## 场景 3: 自定义投掷效果（不影响其他场景）

### 需求：某个场景需要特殊的投掷效果

```gdscript
# scenes/custom_throw_demo.gd
extends Node3D

@onready var dice_manager = $DiceManager

# 自定义投掷参数（只影响这个场景）
@export var custom_force: float = 25.0
@export var custom_angular_force: float = 35.0

func _input(event):
    if event.is_action_pressed("ui_space"):
        throw_with_custom_effect()

func throw_with_custom_effect():
    # 自定义投掷逻辑，不影响其他场景
    for dice in dice_manager.dice_instances:
        var force = Vector3(
            randf_range(-1, 1),
            randf_range(5, 8),
            randf_range(-3, -2)
        ) * custom_force
        
        var angular_force = Vector3(
            randf_range(-custom_angular_force, custom_angular_force),
            randf_range(-custom_angular_force, custom_angular_force),
            randf_range(-custom_angular_force, custom_angular_force)
        )
        
        dice.roll(force, angular_force)
    
    print("使用自定义投掷效果")
```

func get_fallback_color(face_index: int) -> Color:
    var fallback_colors = [
        Color(1, 0, 0, 1),   # 红色
        Color(0, 1, 0, 1),   # 绿色
        Color(0, 0, 1, 1),   # 蓝色
        Color(1, 1, 0, 1),   # 黄色
        Color(1, 0, 1, 1),   # 紫色
