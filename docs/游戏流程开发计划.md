# 游戏流程开发计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 打通从游戏开始到结束的完整流程，实现可玩的游戏循环

**Architecture:** 基于现有战斗框架和关卡系统，创建游戏主入口场景作为流程协调器，通过 LevelStage 管理场景切换，BattleManager 管理战斗，LevelTransitionController 管理关卡转换

**Tech Stack:** Godot 4.x, GDScript, 现有战斗框架单例系统

func load_level_scene(p_node: LevelNode, p_level_data: LevelData) -> bool:
    current_node = p_node
    level_data = p_level_data
    
    # 清理旧场景
    _unload_current_scene()
    
    # 根据节点类型选择场景
    var scene_cfg = scene_config.select_scene_config(p_node.type)
    var scene_resource = load(scene_cfg.scene_path)
    
    current_scene = scene_resource.instantiate()
    add_child(current_scene)
    
    # 使用处理器初始化场景
    var handler = _get_handler_for_node_type(p_node.type)
    if handler:
        handler.initialize_scene(current_scene, p_node, p_level_data)
    
    return true

func _get_handler_for_node_type(type: int) -> SceneHandlerBase:

- [ ] **步骤 2：编写角色选择脚本**

创建 `scenes/ui/character_select/character_select.gd`：
```gdscript
extends Control

signal on_character_selected(hero_ids: Array[int])

@onready var character_container = $MarginContainer/VBoxContainer/CharacterContainer
@onready var confirm_button = $MarginContainer/VBoxContainer/ConfirmButton

var selected_hero_ids: Array[int] = []
var available_heroes: Array[Dictionary] = []

func _ready():
    _load_available_heroes()
    _create_character_cards()

func _load_available_heroes():
    # 从 hero.json 加载所有可用角色（type=1 的玩家角色）
    var file = FileAccess.open("res://table/hero.json", FileAccess.READ)
    if file:
        var json = JSON.new()
        json.parse(file.get_as_text())
        var data = json.get_data()
        
        if data is Dictionary and data.has("heroes"):
            for hero in data["heroes"]:
                if hero.get("type", 1) == 1:  # 只加载玩家角色
                    available_heroes.append(hero)

func _create_character_cards():
    for hero in available_heroes:
        var card = preload("res://scenes/ui/character_select/character_card.tscn").instantiate()
        card.set_hero_data(hero)
        card.on_selected.connect(_on_card_selected.bind(hero))
        character_container.add_child(card)

func _on_card_selected(hero: Dictionary):
    var hero_id = int(hero.get("id", 0))
    
    if hero_id in selected_hero_ids:
        selected_hero_ids.erase(hero_id)
    else:
        if selected_hero_ids.size() < 3:  # 最多选择 3 个角色
            selected_hero_ids.append(hero_id)

func _on_confirm_pressed():
    if selected_hero_ids.size() > 0:
        on_character_selected.emit(selected_hero_ids)
```

extends PanelContainer

signal on_selected(hero: Dictionary)

@onready var portrait = $MarginContainer/VBoxContainer/Portrait

    hero_data = hero
    name_label.text = hero.get("name", "未知角色")
    
    var hp = hero.get("attr_hp", "100")
    var mp = hero.get("attr_mp", "50")
    hp_label.text = "HP: " + str(hp)
    mp_label.text = "MP: " + str(mp)

func _on_gui_input(event):
    if event is InputEventMouseButton and event.pressed and event.button_index == MOUSE_BUTTON_LEFT:
        is_selected = not is_selected
        _update_visual()
        on_selected.emit(hero_data)

func _update_visual():
    if is_selected:
        modulate = Color(0.8, 1.0, 0.8)  # 绿色高亮
    else:
        modulate = Color.WHITE

创建 `scenes/ui/game_clear/game_clear.gd`：
```gdscript
extends Control

@onready var result_label = $MarginContainer/VBoxContainer/ResultLabel
@onready var stats_container = $MarginContainer/VBoxContainer/StatsContainer
@onready var rewards_container = $MarginContainer/VBoxContainer/RewardsContainer
@onready var return_button = $MarginContainer/VBoxContainer/ReturnButton
@onready var restart_button = $MarginContainer/VBoxContainer/RestartButton

var battle_stats: Dictionary = {}
var rewards: Dictionary = {}

func _ready():
    return_button.pressed.connect(_on_return_pressed)
    restart_button.pressed.connect(_on_restart_pressed)

func show_result(victory: bool, stats: Dictionary, rewards_data: Dictionary):
    battle_stats = stats
    rewards = rewards_data
    
    result_label.text = "🎉 胜利！" if victory else "💀 失败..."
    result_label.modulate = Color.GREEN if victory else Color.RED
    
    _show_stats(stats)
    _show_rewards(rewards_data)

func _show_stats(stats: Dictionary):
    # 显示战斗统计
    var turns_label = Label.new()
    turns_label.text = "回合数：%d" % stats.get("turns", 0)
    stats_container.add_child(turns_label)
    
    var damage_dealt_label = Label.new()
    damage_dealt_label.text = "造成伤害：%d" % stats.get("damage_dealt", 0)
    stats_container.add_child(damage_dealt_label)
    
    var damage_received_label = Label.new()
    damage_received_label.text = "受到伤害：%d" % stats.get("damage_received", 0)
    stats_container.add_child(damage_received_label)

func _show_rewards(rewards_data: Dictionary):
    # 显示获得奖励
    var gold_label = Label.new()
    gold_label.text = "获得金币：%d" % rewards_data.get("gold", 0)
    rewards_container.add_child(gold_label)
    
    var exp_label = Label.new()
    exp_label.text = "获得经验：%d" % rewards_data.get("exp", 0)
    rewards_container.add_child(exp_label)

func _on_return_pressed():
    get_tree().change_scene_to_file("res://scenes/ui/title_screen/title_screen.tscn")

func _on_restart_pressed():
    get_tree().change_scene_to_file("res://scenes/game_main/game_main.tscn")
```
